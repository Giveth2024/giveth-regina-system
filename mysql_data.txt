-- 1. Products Table for sellable items
CREATE TABLE products (
    id CHAR(100) NOT NULL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    unit VARCHAR(100) NOT NULL,
    category VARCHAR(100) NOT NULL,
    barcode VARCHAR(150) NULL UNIQUE,
    quantity INT UNSIGNED NOT NULL DEFAULT 0,
    total_units_per_package DECIMAL(10,2) NOT NULL,
    cost_price DECIMAL(10,2) NOT NULL,
    selling_price DECIMAL(10,2) NOT NULL,
    profit DECIMAL(10,2) NOT NULL,
    reorder_level INT UNSIGNED NOT NULL DEFAULT 0,
    image TEXT DEFAULT NULL,
    expiry_date DATE DEFAULT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL
);

-- 2. Raw raw_materials 
For goods we process to products and also we don't sell
CREATE TABLE raw_materials (
    id CHAR(100) NOT NULL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    unit VARCHAR(100) NOT NULL,
    quantity DECIMAL(10,2) UNSIGNED NOT NULL DEFAULT 0.00,
    cost_price DECIMAL(10,2) UNSIGNED NOT NULL,
    reorder_level DECIMAL(10,2) UNSIGNED NOT NULL DEFAULT 0.00,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL
);

-- 3. Purchases table for buying stuff.
CREATE TABLE purchases (
    id CHAR(100) NOT NULL PRIMARY KEY,
    product_id CHAR(36) NOT NULL,
    quantity DECIMAL(10,2) UNSIGNED NOT NULL,
    cost_price DECIMAL(10,2) UNSIGNED NOT NULL,
    total_amount DECIMAL(10,2) UNSIGNED NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_purchases_product
        FOREIGN KEY (product_id)
        REFERENCES products(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);

-- 4. Sales and Sales Item table
CREATE TABLE sales (
    id CHAR(100) NOT NULL PRIMARY KEY,
    invoice_number VARCHAR(50) NOT NULL UNIQUE,
    payment_type VARCHAR(100) NOT NULL,
    total_amount DECIMAL(10,2) UNSIGNED NOT NULL,
    amount_paid DECIMAL(10,2) UNSIGNED NOT NULL,
    change_amount DECIMAL(10,2) UNSIGNED NOT NULL DEFAULT 0.00,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sale_items (
    id CHAR(100) NOT NULL PRIMARY KEY,
    sale_id CHAR(36) NOT NULL,
    product_id CHAR(36) NOT NULL,
    quantity DECIMAL(10,2) UNSIGNED NOT NULL,
    selling_price DECIMAL(10,2) UNSIGNED NOT NULL,
    subtotal DECIMAL(10,2) UNSIGNED NOT NULL,
    profit DECIMAL(10,2) NOT NULL,

    CONSTRAINT fk_sale_items_sale
        FOREIGN KEY (sale_id)
        REFERENCES sales(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    CONSTRAINT fk_sale_items_product
        FOREIGN KEY (product_id)
        REFERENCES products(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT
);

-- Stock table for tracking lifetime of stock
CREATE TABLE stock (
    id CHAR(36) NOT NULL PRIMARY KEY,
    product_id CHAR(36) NOT NULL UNIQUE,
    total_purchases DECIMAL(10,2) UNSIGNED NOT NULL DEFAULT 0.00,
    total_sales DECIMAL(10,2) UNSIGNED NULL DEFAULT 0.00,
    current_quantity DECIMAL(10,2) UNSIGNED NULL DEFAULT 0.00,

    CONSTRAINT fk_stock_product
        FOREIGN KEY (product_id)
        REFERENCES products(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT
);
